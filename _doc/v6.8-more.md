## 主要功能

### 1. 物化視圖 (Materialized View)

```sql
-- 建立物化視圖
CREATE MATERIALIZED VIEW monthly_sales AS
SELECT
    strftime('%Y-%m', order_date) AS month,
    COUNT(*) AS order_count,
    SUM(total) AS revenue
FROM orders
GROUP BY month;

-- 重新整理物化視圖
REFRESH MATERIALIZED VIEW monthly_sales;

-- 刪除物化視圖
DROP MATERIALIZED VIEW monthly_sales;

-- 使用物化視圖（如同普通表）
SELECT * FROM monthly_sales WHERE month = '2026-01';
```

```python
# Python API
conn.execute("""
    CREATE MATERIALIZED VIEW sales_by_region AS
    SELECT region, SUM(sales) FROM orders GROUP BY region
""")

# 重新整理
conn.execute("REFRESH MATERIALIZED VIEW sales_by_region")

# 檢查物化視圖
cursor = conn.execute("SELECT * FROM sqlite_master WHERE type='materialized'")
```

### 2. 進階視窗函數

```sql
-- 移動平均
SELECT
    date,
    price,
    AVG(price) OVER (
        ORDER BY date
        ROWS BETWEEN 2 PRECEDING AND CURRENT ROW
    ) AS moving_avg_3
FROM stock_prices;

-- 首個/最後一個值
SELECT
    department,
    employee_name,
    salary,
    FIRST_VALUE(salary) OVER (PARTITION BY department ORDER BY salary DESC) AS max_salary,
    LAST_VALUE(salary) OVER (PARTITION BY department ORDER BY salary) AS min_salary
FROM employees;

-- NTILE 分桶
SELECT
    score,
    NTILE(4) OVER (ORDER BY score DESC) AS quartile
FROM exam_results;
```

### 3. CTE 增強

```sql
-- 遞迴 CTE（階層查詢）
WITH RECURSIVE org_chart AS (
    -- 根節點
    SELECT id, name, manager_id, 1 AS level
    FROM employees WHERE manager_id IS NULL

    UNION ALL

    -- 遞迴部分
    SELECT e.id, e.name, e.manager_id, oc.level + 1
    FROM employees e
    JOIN org_chart oc ON e.manager_id = oc.id
)
SELECT * FROM org_chart;

-- 多個 CTE
WITH
    active_users AS (
        SELECT user_id, COUNT(*) AS cnt
        FROM sessions
        WHERE last_active > datetime('now', '-7 days')
        GROUP BY user_id
    ),
    orders_summary AS (
        SELECT user_id, SUM(total) AS total
        FROM orders
        GROUP BY user_id
    )
SELECT
    u.name,
    au.cnt AS active_sessions,
    os.total AS lifetime_value
FROM users u
JOIN active_users au ON u.id = au.user_id
JOIN orders_summary os ON u.id = os.user_id;
```

### 5. 批次處理

```python
# 批次插入
from sql6 import BatchOperation

batch = BatchOperation(conn)
batch.insert("users", [
    {"id": 1, "name": "Alice"},
    {"id": 2, "name": "Bob"},
    {"id": 3, "name": "Charlie"},
])
batch.execute()

# 批次更新
batch.update("orders", {"status": "completed"}, "status = 'pending'")
batch.execute()

# 批次刪除
batch.delete("logs", "created_at < datetime('now', '-30 days')")
batch.execute()
```
