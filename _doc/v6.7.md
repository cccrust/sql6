## 主要功能

### 1. 角色型存取控制 (RBAC)

```sql
-- 建立角色
CREATE ROLE reader;
CREATE ROLE writer;
CREATE ROLE admin;

-- 授權
GRANT SELECT ON table_name TO reader;
GRANT INSERT, UPDATE, DELETE ON table_name TO writer;
GRANT ALL ON * TO admin;

-- 撤銷
REVOKE SELECT ON table_name FROM reader;

-- 設定使用者角色
CREATE USER alice WITH PASSWORD 'secret123';
GRANT admin TO alice;

-- 檢查權限
SELECT current_user;
SHOW GRANTS FOR alice;
```

```python
# Python API
from sql6 import RBAC, User, Role, Permission

rbac = RBAC(conn)

# 建立角色
admin_role = rbac.create_role("admin", "管理員角色")
rbac.grant_permission(admin_role, "SELECT", "users")
rbac.grant_permission(admin_role, "INSERT", "orders")

# 建立使用者
alice = rbac.create_user("alice", "password123")
rbac.grant_role(alice, admin_role)

# 權限檢查
if rbac.check_permission("alice", "SELECT", "users"):
    cursor = conn.execute("SELECT * FROM users")
```

### 2. 資料加密

```rust
// 加密配置
struct EncryptionConfig {
    algorithm: EncryptionAlgorithm,  // AES-256, ChaCha20
    key_path: String,              // 金鑰檔案路徑
    mode: EncryptionMode,          //Transparent, Detached
}

enum EncryptionAlgorithm {
    AES256GCM,
    ChaCha20Poly1305,
}

// 透明加密（資料在磁碟上加密）
PRAGMA encryption = aes256gcm;
PRAGMA encryption_key = 'your-256-bit-key';

// 欄位級加密
CREATE TABLE secrets (
    id INTEGER PRIMARY KEY,
    secret_data TEXT ENCRYPT WITH aes256gcm
);

// 金鑰輪換
PRAGMA rotate_encryption_key;
```

```python
# Python 加密 API
from sql6 import Encryption

# 初始化加密
enc = Encryption(conn, algorithm="aes256gcm")

# 加密欄位
encrypted = enc.encrypt_field("sensitive_data", "my-secret-key")

# 解密欄位
decrypted = enc.decrypt_field(encrypted, "my-secret-key")

# 金鑰管理
enc.rotate_key(old_key, new_key)
```

### 3. 審計日誌 (Audit Log)

```sql
-- 啟用審計
PRAGMA audit_log = 1;
PRAGMA audit_log_table = 'audit_log';

-- 審計日誌表
CREATE TABLE audit_log (
    id INTEGER PRIMARY KEY,
    timestamp TEXT,
    user_name TEXT,
    action TEXT,
    object_type TEXT,
    object_name TEXT,
    sql_text TEXT,
    result TEXT
);

-- 查詢審計日誌
SELECT * FROM audit_log
WHERE user_name = 'alice'
AND timestamp > '2026-01-01'
AND action = 'DELETE';
```

```python
# Python 審計 API
from sql6 import AuditLogger

audit = AuditLogger(conn)

# 設定過濾
audit.add_filter(action=["INSERT", "UPDATE", "DELETE"])
audit.add_filter(table=["users", "orders"])

# 自訂審計
audit.log(
    action="CUSTOM",
    details={"change": "salary_update", "old": 5000, "new": 6000}
)

# 查詢審計日誌
logs = audit.query(
    start_time="2026-01-01",
    user_name="alice",
    limit=100
)

# 產生報告
report = audit.generate_compliance_report(
    start="2026-01-01",
    end="2026-01-31"
)
```

### 4. 網路安全

```rust
// TLS/SSL 配置
struct TlsConfig {
    cert_path: String,    // 憑證檔
    key_path: String,     // 私鑰檔
    ca_path: Option<String>,  // CA 憑證（可選）
    verify_client: bool,
}

// 啟用 TLS
let server = Sql6Server::new()
    .with_tls(TlsConfig {
        cert_path: "server.crt",
        key_path: "server.key",
        ..Default::default()
    })
    .listen("0.0.0.0:8080")?;
```

```python
# Python TLS 連線
import sql6

# TLS 連線
conn = sql6.connect(
    "mydb.sql6",
    tls=True,
    tls_cert="client.crt",
    tls_key="client.key"
)

# 或使用 URI
conn = sql6.connect("sql6+tls://alice:pass@host:8080/db")
```
