## 主要功能

### 1. 進階交易控制

```sql
-- Savepoint 支援
BEGIN;
INSERT INTO users VALUES (1, 'Alice');
SAVEPOINT sp1;
INSERT INTO users VALUES (2, 'Bob');
ROLLBACK TO sp1;  -- 只回滾 sp1 之後的變更
COMMIT;  -- 提交 sp1 之前的變更

-- 交易隔離層級
PRAGMA isolation_level = READ COMMITTED;  -- 預設
PRAGMA isolation_level = SERIALIZABLE;

-- 交易狀態查詢
PRAGMA transaction_status;
```


### 3. 備份與還原

```python
# 線上備份（不鎖定）
db.backup("backup.sql6")

# 壓縮備份
db.backup("backup.sql6.gz", compression="gzip")

# 增量備份
db.backup_incremental("incr_20260101.sql6")

# 還原
db.restore("backup.sql6")

# 還原到指定時間點
db.restore("backup.sql6", point_in_time="2026-01-15 10:00:00")
```


### 4. 進階 SQL 功能

```sql
-- 遞迴 CTE
WITH RECURSIVE cte AS (
    SELECT 1 AS n
    UNION ALL
    SELECT n + 1 FROM cte WHERE n < 10
)
SELECT * FROM cte;

-- 視窗函數增強
SELECT
    name,
    department,
    salary,
    AVG(salary) OVER (PARTITION BY department) AS dept_avg,
    RANK() OVER (PARTITION BY department ORDER BY salary DESC) AS dept_rank
FROM employees;

-- Upsert (INSERT ON CONFLICT)
INSERT INTO users (id, name, email)
VALUES (1, 'Alice', 'alice@example.com')
ON CONFLICT (id) DO UPDATE
SET name = EXCLUDED.name, email = EXCLUDED.email;

-- MATERIALIZED 視圖
CREATE MATERIALIZED VIEW monthly_sales AS
SELECT
    strftime('%Y-%m', order_date) AS month,
    SUM(amount) AS total
FROM orders
GROUP BY month;
```

### 5. 監控儀表板

```python
# 即時監控
import sql6.monitor as monitor

# 建立監控器
mon = monitor.DatabaseMonitor("mydb.sql6")

# 即時指標
print(mon.get_current_metrics())

# 歷史趨勢
print(mon.get_trend("queries_per_second", last="1h"))

# 設定告警
mon.set_alert("slow_query", threshold_ms=1000, callback=notify_slack)

# 效能報告
report = mon.generate_report()
report.save("performance_report.html")
```
